\begin{frame}{Deux confusions courantes}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Internet n'est pas le Web}
        \textbf{Internet} est le réseau physique mondial qui relie les ordinateurs.
        \vspace{0.15cm}

        Le \textbf{Web} n'est que l'un des services qui y circulent --- les pages reliées par des \textbf{liens hypertexte} --- au même titre que l'e-mail.
      \end{alertblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Un navigateur n'est pas un moteur de recherche}
        Le \textbf{navigateur} (Firefox, Chrome, Safari) traduit le code reçu pour afficher la page.
        \vspace{0.15cm}

        Le \textbf{moteur de recherche} (Google, Bing) est un site qui aide à trouver des adresses.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \begin{block}{Une conséquence très concrète}
    Chaque navigateur interprète le code à sa manière : l'affichage peut varier de l'un à l'autre. D'où l'habitude à prendre dès maintenant --- \textbf{tester ses pages sur au moins deux navigateurs}.
  \end{block}
\end{frame}

  \end{block}

\begin{frame}{DNS --- du nom à l'adresse}
  Un ordinateur ne comprend pas \texttt{www.hepia.ch} : il lui faut une \textbf{adresse IP} (ex. \texttt{192.0.2.34}). Le \textbf{DNS} traduit le nom de domaine en adresse IP.
  \vspace{0.3cm}
  \begin{block}{Analogie --- L'annuaire téléphonique}
    Tu connais le \textit{nom} (le domaine), l'annuaire te donne le \textit{numéro} (l'IP), et c'est ce numéro qui permet la communication.
  \end{block}
  \vspace{0.2cm}
  \textbf{Étapes simplifiées :}
  \begin{enumerate}
    \item Le navigateur demande : « quelle est l'IP de \texttt{www.hepia.ch} ? »
    \item Le serveur DNS répond avec l'adresse IP
    \item Le navigateur envoie sa requête HTTP à cette adresse
  \end{enumerate}
\end{frame}

      ([yshift=5pt]server.east) -- node[above, font=\scriptsize]{requête SQL} ([yshift=5pt]db.west);
    \draw[-{Stealth[length=5pt]}, thick, warningOrange]
      ([yshift=-5pt]db.west) -- node[below, font=\scriptsize]{données} ([yshift=-5pt]server.east);
  \end{tikzpicture}
  \end{center}
  \vspace{0.3cm}
  \begin{alertblock}{Une règle à retenir dès maintenant}
    Le navigateur ne parle \textbf{jamais} directement à la base de données. Il passe toujours par le serveur. C'est le fondement de la sécurité d'une application web.
  \end{alertblock}
\end{frame}

\begin{frame}{Côté serveur : un choix de langages}
  \begin{block}{Dans ce cours}
    \textbf{Node.js} permet d'exécuter du JavaScript en dehors du navigateur : créer un serveur, recevoir des requêtes HTTP, interroger une base de données, exposer une API.
  \end{block}
  \begin{exampleblock}{Pour information --- d'autres langages, mêmes principes}
    Contrairement au navigateur, un serveur n'est pas limité à un seul langage : \textbf{PHP, Python, Java, Ruby, C\#}\ldots\ coexistent sur le web, et chacun peut jouer le même rôle que Node.js.
    \vspace{0.2cm}

    Le principe reste partout identique : recevoir une requête, dialoguer avec une base de données (en \textbf{SQL}), renvoyer une réponse.
  \end{exampleblock}
\end{frame}

\begin{frame}{Le même langage, deux environnements}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{JavaScript côté client}
        \begin{itemize}
          \item S'exécute dans le \textbf{navigateur}
          \item Accès à la page (le DOM)
          \item Aucun accès au disque
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{JavaScript côté serveur}
        \begin{itemize}
          \item S'exécute avec \textbf{Node.js}
          \item Accès aux fichiers et à la base
          \item Aucun accès à la page
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{0.4cm}
  \begin{center}
    \small Même syntaxe, mêmes règles --- mais deux mondes qu'il ne faut pas confondre.
  \end{center}
\end{frame}

\section{Git --- premiers pas}
% ===============================================================
\partieslide{6. Git --- premiers pas}

\begin{frame}{Pourquoi utiliser Git ?}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item \textbf{Sauvegarder} son travail et son historique
        \item \textbf{Revenir en arrière} en cas d'erreur
        \item \textbf{Collaborer} sans s'écraser
        \item \textbf{Partager} son code (GitHub)
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{alertblock}{Sans Git\ldots}
        \small
        \texttt{index.html}\\
        \texttt{index\_final.html}\\
        \texttt{index\_final2.html}\\
        \texttt{index\_VRAI\_final.html}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.35cm}
  \begin{block}{L'outil standard du métier}
    Git est utilisé par la quasi-totalité des équipes de développement. Dans ce cours, l'\textbf{historique Git fait partie de l'évaluation} : un projet livré en un seul commit n'est pas conforme.
  \end{block}
\end{frame}

\begin{frame}{Les quatre zones de Git}
  \vspace{0.3cm}
  \begin{center}
  \begin{tikzpicture}[font=\small, node distance=1.75cm,
    state/.style={draw=accentBlue, thin, rounded corners=2pt, fill=accentBlue!8,
      minimum width=2.0cm, minimum height=1.15cm, align=center},
    remote/.style={draw=warningOrange, thin, rounded corners=2pt, fill=warningOrange!10,
      minimum width=2.0cm, minimum height=1.15cm, align=center},
    arr/.style={-{Stealth[length=6pt]}, thick}]
    \node[state] (wd) {\textbf{Working}\\{\scriptsize Tes fichiers}};
    \node[state, right=of wd] (st) {\textbf{Stage}\\{\scriptsize Sélection}};
    \node[state, right=of st] (lo) {\textbf{Local}\\{\scriptsize Historique}};
    \node[remote, right=of lo] (gh) {\textbf{GitHub}\\{\scriptsize En ligne}};
    \draw[arr, accentBlue] (wd) -- node[above, font=\scriptsize]{\texttt{git add}} (st);
    \draw[arr, accentBlue] (st) -- node[above, font=\scriptsize]{\texttt{git commit}} (lo);
    \draw[arr, accentBlue] (lo) -- node[above, font=\scriptsize]{\texttt{git push}} (gh);
    \draw[arr, successGreen, bend right=-34] (gh.south) to node[below, font=\scriptsize]{\texttt{git pull}} (lo.south);
  \end{tikzpicture}
  \end{center}
  \begin{exampleblock}{Analogie --- La photo de groupe}
    \textbf{Working} : tout le monde est dans la salle. \textbf{Stage} : certains se placent. \textbf{Commit} : on prend la photo (instantané). \textbf{Push} : on l'envoie sur le serveur partagé.
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Commandes essentielles}
\begin{lstlisting}[language=bash]
git init                    # initialiser un dépôt
git status                  # état des fichiers
git add .                   # tout ajouter au stage
git commit -m "message"     # enregistrer un commit
git push -u origin main     # envoyer sur GitHub (1re fois)
git pull                    # récupérer les changements
\end{lstlisting}
  \vspace{0.2cm}
  \begin{alertblock}{Le bon réflexe}
    Commit \textbf{souvent}, avec des messages clairs. Un commit = une étape logique.
    \vspace{0.1cm}

    \textcolor{successGreen}{\textbf{OUI}} \texttt{Ajout du formulaire de création de sujet} \quad
    \textcolor{alertRed}{\textbf{NON}} \texttt{modif}, \texttt{ça marche}, \texttt{update}
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Le fichier \texttt{.gitignore}}
  Certains fichiers ne doivent \textbf{jamais} être versionnés : mots de passe, dépendances téléchargées, fichiers système.
\begin{lstlisting}[language=bash]
node_modules/     # dépendances Node.js
.env              # mots de passe, clés d'API
.DS_Store         # fichiers système
.vscode/          # réglages locaux de l'éditeur
\end{lstlisting}
  \vspace{0.15cm}
  \begin{alertblock}{À créer avant le premier commit}
    Un fichier déjà commité reste dans l'historique \textbf{même si on l'ajoute au \texttt{.gitignore} ensuite}. C'est ce qui évite, plus tard dans l'année, de publier un mot de passe de base de données sur GitHub.
  \end{alertblock}
\end{frame}
